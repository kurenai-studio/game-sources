## Task 4a: `test_attack_chase` and `test_unreachable_attack`

```gdscript
func test_attack_chase(runner) -> bool:
	var sim = Fixture.world()                          # empty 32x32
	var attacker: int = sim.spawn(1, Vector2(2.5, 2.5), "infantry")
	var target: int = sim.spawn(2, Vector2(12.5, 2.5), "builder")
	runner.check(Fixture.send(sim, {"type": "attack", "entity_ids": [attacker], "target_id": target}).ok, "A1 attack accepted")
	var first_shot: int = -1
	for tick in range(600):
		var before: Array = Fixture.object(sim, attacker).position
		sim.step()
		var after: Array = Fixture.object(sim, attacker).position
		var shots: Array = _shots_by(_events(sim), attacker)
		if not shots.is_empty():
			runner.check(before == after, "A1 never shoots on a tick it moved")
			first_shot = tick
			break
	runner.check(first_shot > 0, "A1 shoots only after closing the distance")
	var attacker_pos: Array = Fixture.object(sim, attacker).position
	var target_pos: Array = Fixture.object(sim, target).position
	runner.check(Vector2(attacker_pos[0], attacker_pos[1]).distance_to(Vector2(target_pos[0], target_pos[1])) <= 4.0 + 1e-6, "A1 fires from inside rifle range")
	runner.check(Vector2i(Vector2(attacker_pos[0], attacker_pos[1]).floor()) != Vector2i(Vector2(target_pos[0], target_pos[1]).floor()), "A1 never enters the target cell")
	# Relocate the target out of range; the attacker must refresh its destination within 15 ticks.
	var old_destination: Variant = Fixture.object(sim, attacker).destination
	runner.check(Fixture.send(sim, {"player": 2, "type": "move", "entity_ids": [target], "target": [12.5, 14.5]}).ok, "A1 target relocation accepted")
	var refreshed: int = -1
	for tick in range(40):
		sim.step()
		var destination: Variant = Fixture.object(sim, attacker).destination
		if destination != null and destination != old_destination:
			refreshed = tick
			break
	runner.check(refreshed != -1 and refreshed <= 16, "A1 pursuit refresh within 15 ticks (plus the tick that applies the command)")
	# Two attackers get distinct destinations and no duplicate queue entries.
	var second: int = sim.spawn(1, Vector2(2.5, 3.5), "infantry")
	runner.check(Fixture.send(sim, {"type": "attack", "entity_ids": [attacker, second], "target_id": target}).ok, "A1 two attackers")
	sim.step()
	var state: Dictionary = sim.snapshot()
	var seen := {}
	for id in state.path_queue:
		runner.check(not seen.has(id), "A1 no duplicate path queue entry")
		seen[id] = true
	var a: Variant = Fixture.object(sim, attacker).destination
	var b: Variant = Fixture.object(sim, second).destination
	runner.check(a == null or b == null or a != b, "A1 distinct pursuit destinations")
	return true

func test_unreachable_attack(runner) -> bool:
	var wall: Array[Vector2i] = []
	for y in range(32):
		wall.append(Vector2i(16, y))
	var sim = Simulation.new(wall, Vector2i(32, 32))
	var attacker: int = sim.spawn(1, Vector2(2.5, 2.5), "infantry")
	var target: int = sim.spawn(2, Vector2(28.5, 2.5), "builder")
	var full_hp: int = Fixture.object(sim, target).hp
	Fixture.send(sim, {"type": "attack", "entity_ids": [attacker], "target_id": target})
	var blocked_tick: int = -1
	var enqueued_after_block: int = 0
	for tick in range(90):
		sim.step()
		var unit: Dictionary = Fixture.object(sim, attacker)
		if blocked_tick == -1 and unit.status == "blocked":
			blocked_tick = tick
		elif blocked_tick != -1 and tick <= blocked_tick + 14 and sim.snapshot().path_queue.has(attacker):
			enqueued_after_block += 1
	runner.check(blocked_tick != -1, "A2 unreachable target ends blocked")
	runner.check(enqueued_after_block == 0, "A2 no path request during the retry interval")
	runner.check(Fixture.object(sim, target).hp == full_hp, "A2 target untouched")
	var none: Array[Vector2i] = []
	runner.check(sim.replace_blocked_cells(none).ok, "A2 wall removed")
	var progressed: bool = false
	for tick in range(90):
		sim.step()
		var unit: Dictionary = Fixture.object(sim, attacker)
		if unit.status == "moving" or unit.status == "path_pending":
			progressed = true
			break
	runner.check(progressed, "A2 opening the wall lets the attack progress at the next retry")
	# Query cap: twenty simultaneous attackers never get more than 8 routes assigned per tick.
	var many := Simulation.new([], Vector2i(32, 32))
	var victim: int = many.spawn(2, Vector2(28.5, 16.5), "builder")
	var ids: Array = []
	for i in range(20):
		ids.append(many.spawn(1, Vector2(2.5 + i % 5, 2.5 + i / 5), "infantry"))
	Fixture.send(many, {"type": "attack", "entity_ids": ids, "target_id": victim})
	for tick in range(30):
		var before := {}
		for entity in many.snapshot().entities:
			before[entity.id] = entity.status
		many.step()
		var assigned: int = 0
		for entity in many.snapshot().entities:
			if before[entity.id] == "path_pending" and entity.status != "path_pending":
				assigned += 1
		runner.check(assigned <= 8, "A2 at most 8 path queries per tick")
	# Stop while queued removes the queue entry.
	var late: int = many.spawn(1, Vector2(1.5, 20.5), "infantry")
	Fixture.send(many, {"type": "attack", "entity_ids": [late], "target_id": victim})
	many.step()
	Fixture.send(many, {"type": "stop", "entity_ids": [late]})
	many.step()
	runner.check(not many.snapshot().path_queue.has(late), "A2 stop removes a queued path")
	return true
```

## Task 5a: `test_mutual_lethal_fire`

```gdscript
func _spawn_duel(first_owner: int, second_owner: int) -> Dictionary:
	var sim = Fixture.world()
	var a: int = sim.spawn(first_owner, Vector2(10.5, 10.5), "tank")
	var b: int = sim.spawn(second_owner, Vector2(14.5, 10.5), "tank")
	return {"sim": sim, "a": a, "b": b}

func test_mutual_lethal_fire(runner) -> bool:
	for owners in [[1, 2], [2, 1]]:                  # second run reverses which owner has the lower ID
		var duel: Dictionary = _spawn_duel(owners[0], owners[1])
		var sim = duel.sim
		var shot_ticks := {duel.a: [], duel.b: []}
		var death_tick: int = -1
		for tick in range(480):
			sim.step()
			var events: Array = _events(sim)
			for event in events:
				if event.type == "shot":
					shot_ticks[event.attacker_id].append(event.tick)
				elif event.type == "death" and death_tick == -1:
					death_tick = event.tick
			if death_tick == tick:
				var kinds: Array = []
				for event in events:
					kinds.append(event.type)
				runner.check(kinds == ["shot", "shot", "death", "death"], "L1 shots first, then deaths")
				runner.check(events[2].object_id < events[3].object_id, "L1 deaths in ascending ID order")
		var expected: Array = [0, 60, 120, 180, 240, 300, 360, 420]
		runner.check(shot_ticks[duel.a] == expected and shot_ticks[duel.b] == expected, "L1 eight shots each, no ninth")
		runner.check(death_tick == 420, "L1 both die in the same fire phase")
		var state: Dictionary = sim.snapshot()
		runner.check(state.entities.is_empty() and state.path_queue.is_empty(), "L1 nothing left behind")
		runner.check(state.next_entity_id == 3, "L1 no IDs consumed by combat")
	# Simultaneous damage to one target aggregates and removes it once.
	var group = Fixture.world()
	for pos in [Vector2(10.5, 10.5), Vector2(10.5, 11.5), Vector2(10.5, 9.5)]:
		group.spawn(1, pos, "tank")
	var victim: int = group.spawn(2, Vector2(14.5, 10.5), "tank")
	var victim_deaths: int = 0
	var victim_death_tick: int = -1
	for tick in range(240):
		group.step()
		for event in _events(group):
			if event.type == "death" and event.object_id == victim:
				victim_deaths += 1
				victim_death_tick = event.tick
	runner.check(victim_deaths == 1 and victim_death_tick == 120, "L1 three volleys kill the target exactly once at tick 120")
	return true
```

## Task 5b: `test_destroyed_producer_accounting`

```gdscript
func test_destroyed_producer_accounting(runner) -> bool:
	var sim = Economy.make_world(10000, Vector2i(32, 32))     # player 1 plus builder ID 1; no supply source, so no income
	sim.add_player(2, 0)
	Economy.build(runner, sim, "reactor", Vector2i(8, 12))
	Economy.build(runner, sim, "depot", Vector2i(8, 5))
	var factory: int = Economy.build(runner, sim, "factory", Vector2i(12, 18))
	runner.check(factory != -1, "P1 factory built")
	var attackers: Array = []
	for pos in [Vector2(20.5, 18.5), Vector2(20.5, 19.5), Vector2(20.5, 20.5), Vector2(19.5, 18.5), Vector2(19.5, 19.5)]:
		attackers.append(sim.spawn(2, pos, "tank"))
	for i in range(3):
		runner.check(Economy.send(sim, {"type": "produce", "building_id": factory, "unit_kind": "tank"}).ok, "P1 tank queued")
	Fixture.send(sim, {"player": 2, "type": "attack", "entity_ids": attackers, "target_id": factory})
	var start: int = sim.snapshot().tick
	var credits_before_death: int = -1
	var credits_after_death: int = -1
	for tick in range(400):
		var state: Dictionary = sim.snapshot()
		var credits: int = state.players[0].credits
		sim.step()
		for event in _events(sim):
			if event.type == "death" and event.object_id == factory:
				credits_before_death = credits
				credits_after_death = sim.snapshot().players[0].credits
		if credits_after_death != -1:
			break
	runner.check(sim.snapshot().tick - 1 == start + 360, "P1 factory dies at the seventh volley")
	runner.check(credits_after_death - credits_before_death == 1800, "P1 two unfinished paid entries refunded once")
	var after: Dictionary = sim.snapshot()
	var tanks: int = 0
	for entity in after.entities:
		if entity.owner == 1 and entity.kind == "tank":
			tanks += 1
	runner.check(tanks == 1, "P1 the tank finished before the death tick survives")
	runner.check(not Fixture.object(sim, factory).has("id"), "P1 producer removed")
	Economy.steps(sim, 60)
	runner.check(sim.snapshot().players[0].credits == credits_after_death, "P1 no second refund")
	return true
```

Notes for the executor:

- Timing logic of `test_destroyed_producer_accounting`: the produce command is applied on the same step as the attack command (tick T0). The first tank finishes after 300 work ticks (spawn at T0+299). Five tanks do 300 000 milliHP per volley against a 2 000 000 milliHP factory, volleys at T0, T0+60, ...; the seventh (T0+360) reaches 2 100 000 and kills it, so exactly the second and third paid entries (2 × 900) are refunded. If your power or production timing differs, recompute these numbers from the code; do not loosen the equality.
- `test_unreachable_attack` relies on the retry interval being 15 ticks (`repath_tick`); if Task 4 pins a different number, change the 14 in `blocked_tick + 14` to interval minus one.
- A destroyed building makes every moving unit re-path for up to 19 ticks (known cost, [envelope](../../reference/m4-scenario-envelope.md)); these tests avoid asserting movement across a death.
