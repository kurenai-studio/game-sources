            if cls.allowed_offhands.is_empty():

            return it.category in cls.allowed_offhands
        return true
    
    Impacto: ✅ Classes sem allowed_weapons agora retornam false ao invés de true

[3] ✅ Loop Infinito Potencial
    Arquivo: scripts/game/dungeon_generator.gd:106
    Correção: Proteção contra edge case (a == b)
    
    ANTES:
    func _carve_line(a: Vector2i, b: Vector2i) -> void:
        var c := a
        var step: Vector2i = Vector2i(signi(b.x - a.x), signi(b.y - a.y))
        while true:
            # ... carving code ...
            if c == b:
                break
            c += step
    
    DEPOIS:
    func _carve_line(a: Vector2i, b: Vector2i) -> void:
        if a == b:
            return
        var c := a
        var step: Vector2i = Vector2i(signi(b.x - a.x), signi(b.y - a.y))
        while true:
            # ... carving code ...
            if c == b:
                break
            c += step
    
    Impacto: ✅ Loop infinito impossível se a == b
